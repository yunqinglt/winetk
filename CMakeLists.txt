cmake_minimum_required(VERSION 3.20)
project(winetk VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

option(WINETK_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Wpedantic -Werror)
    if(WINETK_SANITIZERS)
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address,undefined)
    endif()
endif()
find_package(PkgConfig REQUIRED)
find_package(ZLIB REQUIRED)
pkg_check_modules(PNG REQUIRED IMPORTED_TARGET libpng)

add_library(winetk_core STATIC src/input/parser.c src/terminal/backends/kitty/kitty.c src/session/wine.c)
target_include_directories(winetk_core PUBLIC include)
target_link_libraries(winetk_core PUBLIC ZLIB::ZLIB)
add_executable(winetk-terminal-demo src/demo/main.c src/demo/scene.c src/session/terminal.c src/session/launcher.c)
target_link_libraries(winetk-terminal-demo PRIVATE winetk_core PkgConfig::PNG)

include(CTest)
if(BUILD_TESTING)
    add_executable(test_scale tests/test_scale.c src/demo/scene.c)
    target_include_directories(test_scale PRIVATE include)
    add_test(NAME scale COMMAND test_scale)
    add_executable(test_wine tests/test_wine.c)
    target_link_libraries(test_wine PRIVATE winetk_core)
    add_test(NAME wine COMMAND test_wine)
    add_executable(test_input tests/test_input.c)
    target_link_libraries(test_input PRIVATE winetk_core)
    add_test(NAME input COMMAND test_input)
    add_executable(test_kitty tests/test_kitty.c)
    target_link_libraries(test_kitty PRIVATE winetk_core)
    add_test(NAME kitty COMMAND test_kitty)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME unity_prepare COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_unity_prepare.py)
    add_test(NAME tui COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_tui.py)
    set_tests_properties(tui PROPERTIES TIMEOUT 35)
    add_test(NAME demo_pty COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_demo_pty.py $<TARGET_FILE:winetk-terminal-demo>)
    set_tests_properties(demo_pty PROPERTIES TIMEOUT 35)
endif()
